my-card.js
import { html, css, classify } from 'zilk'

const { WRAP, CARD, LIST } = classify('MyCard')

export default ({ title, list }) => html`
  <section=${WRAP}>
    <article class=${CARD}>
      <h2 class=${CARD.TITLE}>
        ${title}
      </h2>
      <ul class=${LIST}>
        ${list.map(item => html`
          <li class=${LIST.ITEM}>
            ${item}
            </li>
          `)}
      </ul>
    </article>
  </section>
`

export const styles = () => css`
  .${WRAP} {
    position: relative;
    width: 100%;
  }
    .${CARD} {
    background-color: var(--surface);
    padding: 2rem;
  }
    .${CARD.TITLE} {
    color: var(--ink);
    font-weight: bold;
  }
    .${LIST.ITEM} {
    margin-left: 2rem;
  }
`

export const docs = ({ title, list }) => md`
  ## ${title}

  This is some markdown to match the html view

  ${list.map(item => md`
- ${item}
    `)}
`
